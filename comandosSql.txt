-- Habilita extensão para UUID se necessário (opcional, mas recomendado)
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- 1. Tabela USUARIO
CREATE TABLE usuario (
    id UUID NOT NULL PRIMARY KEY,
    username VARCHAR(200) UNIQUE,
    password VARCHAR(200),
    email VARCHAR(200),
    roles VARCHAR[] -- Array de strings para o PostgreSQL
);

-- 2. Tabela AUTOR
CREATE TABLE autor (
    id UUID NOT NULL PRIMARY KEY,
    nome VARCHAR(200) NOT NULL,
    data_nascimento DATE NOT NULL,
    nacionalidade VARCHAR(100) NOT NULL,
    data_cadastro TIMESTAMP,
    data_atualizacao TIMESTAMP,
    id_usuario UUID REFERENCES usuario(id) -- Relacionamento com Usuario
);

-- 3. Tabela LIVRO
CREATE TABLE livro (
    id UUID NOT NULL PRIMARY KEY,
    isbn VARCHAR(30) NOT NULL UNIQUE,
    titulo VARCHAR(150) NOT NULL,
    data_publicacao DATE NOT NULL,
    genero VARCHAR(30) NOT NULL,
    preco NUMERIC(18, 2),
    data_cadastro TIMESTAMP,
    data_atualizacao TIMESTAMP,
    id_autor UUID NOT NULL REFERENCES autor(id), -- Foreign Key para Autor
    id_usuario UUID REFERENCES usuario(id),      -- Foreign Key para Usuario

    -- Constraint para garantir que o gênero seja válido (opcional, mas seguro)
    CONSTRAINT chk_genero CHECK (genero IN ('FICCAO', 'FANTASIA', 'MISTERIO', 'ROMANCE', 'BIOGRAFIA', 'CIENCIA'))
);

-- 4. Tabela CLIENT (para OAuth2)
CREATE TABLE client (
    id UUID NOT NULL PRIMARY KEY,
    client_id VARCHAR(200),
    client_secret VARCHAR(600),
    redirect_uri VARCHAR(300),
    scope VARCHAR(100)
);
